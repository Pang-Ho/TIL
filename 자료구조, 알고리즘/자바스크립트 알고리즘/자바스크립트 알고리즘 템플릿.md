// 퀵 정렬
function quickSort(arr, left = 0, right = arr.length - 1) {
    if (left < right) {
        const pivot = partition(arr, left, right);
        quickSort(arr, left, pivot - 1);
        quickSort(arr, pivot + 1, right);
    }
    return arr;
}

function partition(arr, left, right) {
    const pivot = arr[right];
    let i = left - 1;
    
    for (let j = left; j < right; j++) {
        if (arr[j] <= pivot) {
            i++;
            [arr[i], arr[j]] = [arr[j], arr[i]];
        }
    }
    [arr[i + 1], arr[right]] = [arr[right], arr[i + 1]];
	return i + 1;
}

// 병합 정렬
function mergeSort(arr) {
    if (arr.length <= 1) return arr;
    
    const mid = Math.floor(arr.length / 2);
    const left = mergeSort(arr.slice(0, mid));
    const right = mergeSort(arr.slice(mid));

	return merge(left, right);
}

function merge(left, right) {
    const result = [];
    let i = 0, j = 0;
    
    while (i < left.length && j < right.length) {
        if (left[i] <= right[j]) {
            result.push(left[i++]);
        } else {
            result.push(right[j++]);
        }
    }

    return result.concat(left.slice(i)).concat(right.slice(j));
}

// 이진 탐색
function binarySearch(arr, target) {
    let left = 0, right = arr.length - 1;
    

while (left <= right) {
    const mid = Math.floor((left + right) / 2);
    if (arr[mid] === target) return mid;
    else if (arr[mid] < target) left = mid + 1;
    else right = mid - 1;
}
return -1;

}

    visited.add(start);
    

while (queue.length > 0) {
    const node = queue.shift();

            queue.push(neighbor);
        }
    }
}

}

// ========== 3. 동적 계획법 (DP) ==========

// 피보나치 수열
function fibonacci(n) {
    const dp = [0, 1];
    for (let i = 2; i <= n; i++) {
        dp[i] = dp[i-1] + dp[i-2];
    }
    return dp[n];
}

// 최장 증가 부분 수열 (LIS)
function lis(arr) {
    const dp = new Array(arr.length).fill(1);
    

for (let i = 1; i < arr.length; i++) {
    for (let j = 0; j < i; j++) {
        if (arr[j] < arr[i]) {
            dp[i] = Math.max(dp[i], dp[j] + 1);
        }
    }
}

return Math.max(...dp);

}

// 배낭 문제
function knapsack(weights, values, capacity) {
    const n = weights.length;
    const dp = Array.from({length: n + 1}, () => new Array(capacity + 1).fill(0));
    

for (let i = 1; i <= n; i++) {
    for (let w = 1; w <= capacity; w++) {
        if (weights[i-1] <= w) {
            dp[i][w] = Math.max(
                dp[i-1][w],
                dp[i-1][w - weights[i-1]] + values[i-1]
            );
        } else {
            dp[i][w] = dp[i-1][w];
        }
    }
}

return dp[n][capacity];

}
```

// KMP 알고리즘
function kmp(text, pattern) {
    const lps = computeLPS(pattern);
    const result = [];
    let i = 0, j = 0;
    

while (i < text.length) {
    if (text[i] === pattern[j]) {
        i++;
        j++;
    }
    
    if (j === pattern.length) {
        result.push(i - j);
        j = lps[j - 1];
    } else if (i < text.length && text[i] !== pattern[j]) {
        if (j !== 0) {
            j = lps[j - 1];
        } else {
            i++;
        }
    }
}

return result;

}

function computeLPS(pattern) {
    const lps = new Array(pattern.length).fill(0);
    let len = 0;
    let i = 1;
    

while (i < pattern.length) {
    if (pattern[i] === pattern[len]) {
        len++;
        lps[i] = len;
        i++;
    } else {
        if (len !== 0) {
            len = lps[len - 1];
        } else {
            lps[i] = 0;
            i++;
        }
    }
}

return lps;

}

// Rolling Hash 
function rollingHashSearch(text, pattern) {
    const base = 256;
    const mode = 2_147_483_647;
    const n = text.length;
    const m = pattern.length;
    
    let tHash = 0;
    let pHash = 0;
    let h = 1;
    
    for (let i = 0; i < m - 1; i++) h = (h * base) % mod;
    for (let i = 0; i < m; i++) {
        tHash = (tHash * base + text.charCodeAt(i)) % mod;
        pHash = (pHash * base + pattern.charCodeAt(i)) % mod;
    }
    
    for (let i = 0; i <= n - m; i++) {
        if (tHash === pHash) {
            if (text.substr(i, m) === pattern) {
                return i;
            }
        }
        
        if (i < n - m) {
            tHash = (tHash - h * text.charCodeAt(i)) % mod;
            tHash = (tHash * base + text.charCodeAt(i + m)) % mod;
            if (tHash < 0) tHash += mod;
        }
    }
    return -1;
}

// ========== 6. 그래프 알고리즘 ==========

// 다익스트라 알고리즘
function dijkstra(graph, start) {
    const distances = {};
    const visited = new Set();
    const pq = [[0, start]]; // [distance, node]
    

// 초기화
for (const node in graph) {
    distances[node] = Infinity;
}
distances[start] = 0;

while (pq.length > 0) {
    pq.sort((a, b) => a[0] - b[0]);
    const [currentDist, currentNode] = pq.shift();
    
    if (visited.has(currentNode)) continue;
    visited.add(currentNode);
    
    for (const [neighbor, weight] of graph[currentNode]) {
        const distance = currentDist + weight;
        
        if (distance < distances[neighbor]) {
            distances[neighbor] = distance;
            pq.push([distance, neighbor]);
        }
    }
}

return distances;

}

// 플로이드-워셜 알고리즘
function floydWarshall(graph) {
    const n = graph.length;
    const dist = graph.map(row => [...row]);
    

for (let k = 0; k < n; k++) {
    for (let i = 0; i < n; i++) {
        for (let j = 0; j < n; j++) {
            if (dist[i][k] + dist[k][j] < dist[i][j]) {
                dist[i][j] = dist[i][k] + dist[k][j];
            }
        }
    }
}

return dist;

}

// 최대공약수 (GCD)
function gcd(a, b) {
    while (b !== 0) {
        [a, b] = [b, a % b];
    }
    return a;
}

// 최소공배수 (LCM)
function lcm(a, b) {
    return (a * b) / gcd(a, b);
}

// 소수 판별
function isPrime(n) {
    if (n < 2) return false;
    for (let i = 2; i <= Math.sqrt(n); i++) {
        if (n % i === 0) return false;
    }
    return true;
}

// 에라토스테네스의 체
function sieveOfEratosthenes(n) {
    const primes = new Array(n + 1).fill(true);
    primes[0] = primes[1] = false;
    

for (let i = 2; i * i <= n; i++) {
    if (primes[i]) {
        for (let j = i * i; j <= n; j += i) {
            primes[j] = false;
        }
    }
}

return primes.map((isPrime, num) => isPrime ? num : null).filter(num => num !== null);

}

function twoSum(nums, target) {
    const map = new Map();
    

for (let i = 0; i < nums.length; i++) {
    const complement = target - nums[i];

function isPalindrome(s) {
    const cleaned = s.toLowerCase().replace(/[^a-z0-9]/g, '');
    let left = 0, right = cleaned.length - 1;
    

while (left < right) {
    if (cleaned[left] !== cleaned[right]) {
        return false;
    }
    left++;
    right--;
}

return true;

}

// 예시 4: 괄호 유효성 검사
function isValidParentheses(s) {
    const stack = [];
    const pairs = {'(': ')', '[': ']', '{': '}'};
    

for (const char of s) {
    if (char in pairs) {
        stack.push(char);
    } else {
        const last = stack.pop();
        if (pairs[last] !== char) {
            return false;
        }
    }
}

return stack.length === 0;

}

// 예시 7: 순열 생성
function permute(nums) {
    const result = [];
    

function backtrack(current) {
    if (current.length === nums.length) {
        result.push([...current]);
        return;
    }
    
    for (const num of nums) {
        if (!current.includes(num)) {
            current.push(num);
            backtrack(current);
            current.pop();
        }
    }
}

backtrack([]);
return result;

}

        result.push([...current]);
        return;
    }
    
    for (let i = start; i <= n; i++) {
        current.push(i);
        backtrack(i + 1, current);
        current.pop();
    }
}

backtrack(1, []);
return result;

}

// ========== 자주 사용하는 패턴 ==========
// 슬라이딩 윈도우
function maxSumSubarray(arr, k) {
    let maxSum = 0;
    let windowSum = 0;
    

// 첫 번째 윈도우
for (let i = 0; i < k; i++) {
    windowSum += arr[i];
}
maxSum = windowSum;

// 슬라이딩
for (let i = k; i < arr.length; i++) {
    windowSum = windowSum - arr[i - k] + arr[i];
    maxSum = Math.max(maxSum, windowSum);
}

return maxSum;

}

// 투 포인터
function removeDuplicates(nums) {
    if (nums.length === 0) return 0;
    

let slow = 0;
for (let fast = 1; fast < nums.length; fast++) {
    if (nums[fast] !== nums[slow]) {
        slow++;
        nums[slow] = nums[fast];
    }
}

return slow + 1;

}

// 백트래킹 템플릿
function backtrackTemplate(candidates, target) {
    const result = [];
    

function backtrack(start, current, remaining) {
    // 기저 조건
    if (remaining === 0) {
        result.push([...current]);
        return;
    }
    
    if (remaining < 0) return;
    
    // 선택지 탐색
    for (let i = start; i < candidates.length; i++) {
        current.push(candidates[i]);
        backtrack(i, current, remaining - candidates[i]);
        current.pop(); // 백트래킹
    }
}

backtrack(0, [], target);
return result;

}
```
